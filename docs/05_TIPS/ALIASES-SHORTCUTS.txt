
## Env Variables for Speed

```bash
# ~/.zshrc additions
export PENTEST="$HOME/pentest"
export TOOLS="/opt/tools"
export WORDLISTS="/usr/share/wordlists"
export DOWNLOADS="$HOME/Downloads"

# Quick access function
pentest() { cd "$PENTEST/$1" }
tools() { cd "$TOOLS/$1" }
```

## Functions for Speed

```bash
# Quick scan
scan() { nmap -sV -sC -p- "$1" -oN scan-"$1".nmap; }

# Quick web scan
web() { nikto -h "$1" | tee nikto-"$1".txt; }

# Extract archives
extract() { for file in "$@"; do 7z x "$file"; done; }

# Quick http server
serve() { python3 -m http.server "$1" -d "${2:-.}"; }

# URL encode
urlencode() { python3 -c "import urllib.parse; print(urllib.parse.quote('$1'))"; }

# Base64 encode/decode
b64e() { echo "$1" | base64; }
b64d() { echo "$1" | base64 -d; }

# Hash file
hash() { echo "$1:$(sha256sum "$1" | cut -d' ' -f1)"; }
```

## Tmux Config

```bash
# ~/.tmux.conf
set -g prefix C-a
unbind C-b
bind C-a send-prefix

# Split
bind | split-window -h
bind - split-window -v

# Navigation
bind h select-pane -L
bind j select-pane -D
bind k select-pane -U
bind l select-pane -R

# Resize
bind -r H resize-pane -L 5
bind -r J resize-pane -D 5
bind -r K resize-pane -U 5
bind -r L resize-pane -R 5

# Status
set -g status-style bg=colour235
set -g status-left "#S"
set -g status-right "%H:%M %d-%b-%y"
```
