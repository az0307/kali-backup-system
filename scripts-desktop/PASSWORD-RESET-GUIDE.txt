# ================================================
# PASSWORD RESET GUIDE - Windows SAM Database
# ================================================
# Run these commands in Kali Live terminal

## QUICK RESET (Administrator):

sudo su
fdisk -l                                    # Find Windows partition (usually /dev/sda1)
mount /dev/sda1 /mnt                         # Mount Windows
cd /mnt/Windows/System32/config
chntpw -i SAM

# Menu options:
# 1 = Edit user data and password
#   Then select username
#   1 = Clear password (blank)
# q = Quit
# y = Save

# Then:
cd /
umount /mnt
reboot
